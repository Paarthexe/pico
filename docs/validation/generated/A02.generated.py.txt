import pandas as pd
from scipy.stats import chi2_contingency
from pico_tools import load_dataset, save_result

# Load dataset
df = load_dataset()

# Handle missing values
df.dropna(subset=['income', 'purchases', 'churn'], inplace=True)

# Create quartiles for income and purchases
df['income_quartile'] = pd.qcut(df['income'], q=4, labels=False)
df['purchases_quartile'] = pd.qcut(df['purchases'], q=4, labels=False)

# Create contingency table
contingency_table = pd.crosstab(df['purchases_quartile'], df['churn'])

# Perform chi-square test
chi2, p, dof, expected = chi2_contingency(contingency_table)

# Prepare result
n = len(df)
conclusion = "Purchase frequency quartiles are associated with churn rates."
limitations = [
    "Independent/exchangeable observations are assumed; the app cannot establish this from a CSV.",
    "7 duplicate rows may violate independence.",
    "Used 2235/2400 observations; missingness/subsetting can bias the result."
]
table = contingency_table.to_dict(orient='records')
statistic = chi2
p_value = p
effect = None
method = "Chi-square test"
assumptions = [
    "Independence of observations",
    "Expected cell counts >= 5"
]
diagnostics = {
    "min_expected_count": expected.min()
}

# Save result
save_result(n=n, conclusion=conclusion, limitations=limitations, table=table, statistic=statistic, p_value=p_value, effect=effect, method=method, assumptions=assumptions, diagnostics=diagnostics)
