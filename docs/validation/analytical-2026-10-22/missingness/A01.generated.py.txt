from pico_tools import load_dataset, save_result
import pandas as pd
import scipy.stats as stats

# Load the dataset
df = load_dataset()

# Pairwise complete cases
paired = df[['x', 'y']].dropna()

# Apply qcut for numeric x
paired['x'] = pd.qcut(paired['x'], 4, labels=False)

# Create contingency table
counts = pd.crosstab(paired['x'].astype(str), paired['y'].astype(str))

# Perform Pearson chi-squared test
result = stats.chi2_contingency(counts, correction=False)

# Prepare the table for the result
table = [
    {'group': str(g), 'category': str(c), 'count': int(counts.loc[g, c])}
    for g in counts.index for c in counts.columns
]

# Prepare diagnostics
diagnostics = {
    'verification_test': 'pearson_chi2',
    'verification_missing': 'complete_case',
    'verification_bins': 4
}

# Save the result
save_result(
    n=len(paired),
    table=table,
    statistic=float(result.statistic),
    p_value=float(result.pvalue),
    effect=None,
    conclusion="Insufficient evidence to claim a significant association between x and y.",
    limitations=["Missingness in x and y was handled by pairwise complete cases.", "Q-cut was applied to x for categorization.", "No imputation was performed."],
    method="Pearson chi-squared test",
    assumptions=["Independence of observations.", "Expected cell counts in the contingency table are at least 5."],
    diagnostics=diagnostics
)
